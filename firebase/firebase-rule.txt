rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function isOwner(field) { return signedIn() && resource.data[field] == request.auth.uid; }
    function createsOwn(field) { return signedIn() && request.resource.data[field] == request.auth.uid; }
    function isAdmin() { return signedIn() && request.auth.token.role == 'admin'; }

    match /users/{userId} {
      allow create: if signedIn() && request.auth.uid == userId
        && request.resource.data.role == 'doctor';
      allow read: if signedIn() && (request.auth.uid == userId || isAdmin());
      allow update: if signedIn() && request.auth.uid == userId
        && request.resource.data.role == resource.data.role;
      allow delete: if isAdmin();
    }
    match /diagnosis/{id} {
      allow create: if createsOwn('createdBy');
      allow read: if isAdmin() || isOwner('createdBy');
      allow update, delete: if isAdmin();
    }
    match /patients/{id} {
      allow create: if createsOwn('createdBy');
      allow read: if isAdmin() || isOwner('createdBy');
      allow update: if isOwner('createdBy') && request.resource.data.createdBy == resource.data.createdBy;
      allow delete: if isAdmin();
    }
    match /chat_history/{id} {
      allow create: if createsOwn('uid');
      allow read, delete: if isOwner('uid');
      allow update: if false;
    }
    match /model_metrics/{id} {
      allow read: if isAdmin();
      allow write: if false;
    }
  }
}
